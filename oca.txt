#!/usr/bin/env bash
# oca — OpenCode Agent launcher for Termux
# https://github.com/YOURUSER/oca-termux
set -euo pipefail

VERSION="1.0.0"

C_G='\033[1;32m'; C_C='\033[1;36m'; C_Y='\033[1;33m'; C_R='\033[1;31m'; C_D='\033[2m'; C_0='\033[0m'

banner() {
  printf "${C_C}"
  cat <<'ASCII'
   ____   ____  _
  / __ \ / ___|/ \
 | |  | | |   / _ \
 | |__| | |__/ ___ \
  \____/ \____/_/  \_\
      A G E N T · tmux/termux
ASCII
  printf "${C_0}${C_D}  oca v%s — cd, ask, ship.${C_0}\n\n" "$VERSION"
}

die() { printf "${C_R}✗ %s${C_0}\n" "$1" >&2; exit 1; }
info() { printf "${C_C}➜ %s${C_0}\n" "$1"; }
ok() { printf "${C_G}✔ %s${C_0}\n" "$1"; }
warn() { printf "${C_Y}! %s${C_0}\n" "$1"; }

need_opencode() {
  command -v opencode >/dev/null 2>&1 || die "opencode not found. Run: oca setup"
}

# ---- doctor: report on toolchain / PATH -----------------------------------
doctor() {
  banner
  info "Environment check"
  echo "  termux prefix : ${PREFIX:-N/A}"
  echo "  shell         : $SHELL"
  echo "  pwd           : $(pwd)"
  echo

  info "Core tools"
  for bin in opencode node npm git bash curl rg fzf python; do
    if command -v "$bin" >/dev/null 2>&1; then
      ok "$bin  -> $(command -v "$bin")"
    else
      warn "$bin  -> not in PATH"
    fi
  done
  echo

  if command -v pkg >/dev/null 2>&1; then
    info "Installed pkg packages (top 15)"
    pkg list-installed 2>/dev/null | tail -n +2 | head -15 | sed 's/^/  /'
  else
    warn "pkg not available (not running in Termux?)"
  fi
  echo

  info "Everything reachable via \$PATH:"
  echo "$PATH" | tr ':' '\n' | sed 's/^/  /'
}

# ---- setup: install opencode + common deps ---------------------------------
setup() {
  banner
  if command -v pkg >/dev/null 2>&1; then
    info "Updating pkg and installing base deps (nodejs, git, curl, ripgrep)"
    pkg update -y && pkg install -y nodejs git curl ripgrep
  else
    warn "pkg not found — skipping Termux package install"
  fi

  if command -v opencode >/dev/null 2>&1; then
    ok "opencode already installed: $(opencode --version 2>/dev/null || echo unknown)"
  else
    info "Installing opencode via npm"
    npm i -g opencode-ai@latest || die "npm install failed. Try: curl -fsSL https://opencode.ai/install | bash"
    ok "opencode installed"
  fi

  echo
  ok "Setup complete. Run: opencode auth   (to add your model provider key)"
  ok "Then: cd into a project and run 'oca'"
}

usage() {
  banner
  cat <<EOF
Usage:
  oca                          launch the agent (TUI) in the current folder
  oca "do the thing"           one-shot: run a prompt against current folder
  oca <path>                   cd into <path>, then launch the agent (TUI)
  oca <path> "do the thing"    cd into <path>, then run a one-shot prompt

  oca doctor                   show installed tools, PATH, pkg packages
  oca setup                    install opencode + base deps via pkg/npm
  oca -h | --help              show this help
  oca -v | --version           show oca version
EOF
}

# ---- main --------------------------------------------------------------
main() {
  case "${1:-}" in
    ""|-h|--help|help) [ -z "${1:-}" ] || { usage; exit 0; } ;;
  esac

  case "${1:-}" in
    doctor) doctor; exit 0 ;;
    setup)  setup; exit 0 ;;
    -h|--help) usage; exit 0 ;;
    -v|--version) echo "oca $VERSION"; exit 0 ;;
  esac

  local target_dir="."
  local prompt=""

  if [ -n "${1:-}" ] && [ -d "$1" ]; then
    target_dir="$1"
    shift
    prompt="$*"
  else
    prompt="$*"
  fi

  cd "$target_dir" || die "no such directory: $target_dir"
  need_opencode

  banner
  info "project: $(pwd)"
  if [ -d .git ]; then
    branch=$(git branch --show-current 2>/dev/null || echo "?")
    info "git branch: $branch"
  fi
  echo

  if [ -n "$prompt" ]; then
    info "running one-shot prompt"
    exec opencode run "$prompt"
  else
    info "launching interactive agent (Ctrl+C to quit)"
    exec opencode
  fi
}

main "$@"
